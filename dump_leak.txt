   112			return (ERR_MALLOC);

   114			return (ERR_MALLOC);

   116			return (ERR_MALLOC);

    19	int	parse_farm(t_farm *farm)
    20	{
    21		char	*line;
    22		int		err;
    23		int		cmd;
    24	
    25		err = ERR_NO_ERROR;
    26		cmd = PARSE_CMD_NONE;
    27		line = get_next_line(0);
    28		while (!err && parse_ants(&err, &line, farm))
    29			read_next_line(&line);
    30		while (!err && parse_rooms(&err, &line, farm, &cmd))
    31			read_next_line(&line);
    32		if (!err && cmd != PARSE_CMD_NONE)
    33			err = ERR_ROOM;
    34		while (!err && parse_links(&err, &line, farm))
    35			read_next_line(&line);
    36		if (!err && line)
    37			err = ERR_FORMAT;
    38		if (!err)
    39			err = parse_check_farm(farm);
    40		if (line)
    41			free(line);
    42		farm->error = err;
    43		return (err);
    44	}

    22	int	parse_rooms(int *err, char **line, t_farm *farm, int *cmd) {
    23		if (*err || !*line)
    24			return (0);
    25		if (parse_is_start_cmd(*line) || parse_is_end_cmd(*line))
    26			return (handle_room_cmd(err, line, farm, cmd));
    27		if (parse_is_comment(*line))
    28		{
    29			if (*cmd != PARSE_CMD_NONE)
    30				*err = ERR_ROOM;
    31			parse_consume_line(line);
    32			return (1);
    33		}
    34		if (parse_count_char(*line, ' ') == 0)
    35		{
    36			if (*cmd != PARSE_CMD_NONE)
    37			{
    38				*err = ERR_ROOM;
    39				parse_consume_line(line);
    40				return (1);
    41			}
    42			return (0);
    43		}
    44		*err = parse_room_line(farm, *line, *cmd);
    45		*cmd = PARSE_CMD_NONE;
    46		*line = NULL;
    47		return (1);
    48	}

    50	static int	handle_room_cmd(int *err, char **line, t_farm *farm, int *cmd) {
    51		if (*cmd != PARSE_CMD_NONE)
    52			*err = ERR_ROOM;
    53		else if (parse_is_start_cmd(*line))
    54		{
    55			if (farm->start)
    56				*err = ERR_START_END;
    57			else
    58				*cmd = PARSE_CMD_START;
    59		}
    60		else
    61		{
    62			if (farm->end)
    63				*err = ERR_START_END;
    64			else
    65				*cmd = PARSE_CMD_END;
    66		}
    67		parse_consume_line(line);
    68		return (1);
    69	}

    71	static int	parse_room_line(t_farm *farm, char *line, int cmd) {
    72		t_room_parse	info;
    73		int				err;
    74	
    75		err = fill_room_parse(line, &info);
    76		if (!err)
    77			err = validate_room_parse(farm, &info);
    78		if (!err)
    79			err = create_room(farm, &info, cmd);
    80		if (err)
    81			free(line);
    82		return (err);
    83	}
    84	
    85	static int	fill_room_parse(char *line, t_room_parse *info) {
    86		char	*first;
    87		char	*second;
    88	
    89		if (parse_count_char(line, ' ') != 2)
    90			return (ERR_ROOM);
    91		first = ft_strchr(line, ' ');
    92		second = ft_strchr(first + 1, ' ');
    93		*first = '\0';
    94		*second = '\0';
    95		info->name = line;
    96		info->x_str = first + 1;
    97		info->y_str = second + 1;
    98		parse_strip_newline(info->y_str);
    99		return (ERR_NO_ERROR);
   100	}

   102	static int	validate_room_parse(t_farm *farm, t_room_parse *info) {
   103		char	*coord_key;
   104	
   105		if (!info->name[0] || !info->x_str[0] || !info->y_str[0])
   106			return (ERR_ROOM);
   107		if (info->name[0] == 'L' || info->name[0] == '#')
   108			return (ERR_ROOM);
   109		if (ft_strchr(info->name, '-'))
   110			return (ERR_ROOM);
   111		if (custom_atoi(info->x_str, &info->x))
   112			return (ERR_ROOM);
   113		if (custom_atoi(info->y_str, &info->y))
   114			return (ERR_ROOM);
   115		coord_key = make_coord_key(info->x, info->y);
   116		if (!coord_key)
   117			return (ERR_MALLOC);
   118		if (hashmap_get(farm->rooms_by_name, info->name)
   119			|| hashmap_get(farm->rooms_by_coord, coord_key))
   120		{
   121			free(coord_key);
   122			return (ERR_ROOM);
   123		}
   124		free(coord_key);
   125		return (ERR_NO_ERROR);
   126	}

   129		t_room	*room;

   133		if (!room)
   134			return (ERR_MALLOC);
   135		room->id = (int)farm->rooms.count;
   136		room->name = info->name;
   137		room->x = info->x;
   138		room->y = info->y;
   139		if (vector_init(&room->neighbors, 4))
   140		{
   141			free(room);
   142			return (ERR_MALLOC);
   143		}
   144		err = farm_add_room(farm, room);
   145		if (err)
   146		{
   147			room_destroy(room);
   148			return (err);
   149		}
   150		if (cmd == PARSE_CMD_START)
   151			farm->start = room;
   152		else if (cmd == PARSE_CMD_END)
   153			farm->end = room;
   154		return (ERR_NO_ERROR);
   155	}

    18	int	custom_atoi(const char *str, int *out) {
    19		long	result;
    20		int		sign;
    21	
    22		result = 0;
    23		sign = 1;
    24		if (*str == '-' || *str == '+')
    25		{
    26			if (*str == '-')
    27				sign = -1;
    28			str++;
    29		}
    30		if (!*str)
    31			return (1);
    32		while (*str)
    33		{
    34			if (!ft_isdigit(*str))
    35				return (1);
    36			result = result * 10 + (*str - '0');
    37			if ((sign == 1 && result > INT_MAX)
    38				|| (sign == -1 && result > -(long)INT_MIN))
    39				return (1);
    40			str++;
    41		}
    42		*out = (int)(result * sign);
    43		return (0);
    44	}

    55		return (out);

    76	}

    18		if (!vec->data)

    34	int	vector_push_back(t_vector *vec, void *elem)
    35	{
    36		void	**new_data;
    37		size_t	new_capacity;
    38		size_t	i;
    39	
    40		if (vec->count >= vec->capacity)
    41		{
    42			new_capacity = vec->capacity * 2;
    43			new_data = malloc(new_capacity * sizeof(void *));
    44			if (!new_data)
    45				return (-1);
    46			i = 0;
    47			while (i < vec->count)
    48			{
    49				new_data[i] = vec->data[i];
    50				i++;
    51			}
    52			free(vec->data);
    53			vec->data = new_data;
    54			vec->capacity = new_capacity;
    55		}
    56		vec->data[vec->count++] = elem;
    57		return (0);
    58	}
    59	/***
    60	 * @brief Remove an element from the vector at the specified index
    61	 * @param vec The vector to remove from
    62	 * @param index The index of the element to remove
    63	 */
    64	void vector_remove(t_vector *vec, size_t index)
    65	{
    66		int i;
    67	
    68		if (!vec || index >= vec->count)
    69			return;
    70		i = index;
    71		while (i < vec->count - 1)
    72		{
    73			vec->data[i] = vec->data[i + 1];
    74			i++;
    75		}
    76		vec->count--;
    77	}

    17	 * @brief Create a new hashmap with the specified initial capacity.
    18	 */
    19	t_hashmap	*hashmap_create(size_t initial_capacity)
    20	{
    21		t_hashmap	*map;
    22	
    23		map = (t_hashmap *)malloc(sizeof(t_hashmap));
    24		if (!map)
    25			return (NULL);
    26		map->nodes = (t_hashnode *)ft_calloc(initial_capacity, sizeof(t_hashnode));
    27		if (!map->nodes)
    28		{
    29			free(map);
    30			return (NULL);
    31		}
    32		map->capacity = initial_capacity;
    33		map->count = 0;
    34		return (map);
    35	}
    36	
    37	/***
    38	 * @brief destroy the hashmap and free all its nodes
    39	 * @param map the hashmap to destroy
    40	 * @param destroy a function to destroy the value of each node
    41	 * @note if destroy is NULL, the value will not be destroyed
    42	 */
    43	void	hashmap_destroy(t_hashmap *map, void (*destroy)(void *))
    44	{
    45		size_t	i;
    46	
    47		if (!map)
    48			return ;
    49		if (map->nodes)
    50		{
    51			i = 0;
    52			while (i < map->capacity)
    53			{
    54				if (map->nodes[i].key)
    55				{
    56					free(map->nodes[i].key);
    57					if (destroy && map->nodes[i].value)
    58						destroy(map->nodes[i].value);
    59				}
    60				i++;
    61			}
    62			free(map->nodes);
    63		}
    64		free(map);
    65	}

    19	static void	swap_nodes(t_hashnode *a, t_hashnode *b);
    20	int			robinhood_probe(t_hashmap *map, t_hashnode new_node);
    21	int			hashmap_rehash(t_hashmap *map);
    22	
    23	/***
    24	 * @brief Insert a key-value pair into the hashmap
    25	 * @param map the hashmap to insert into
    26	 * @param key the key to insert
    27	 * @param value the value to insert
    28	 * @return 0 on success, -1 on failure
    29	*/
    30	int	hashmap_insert(t_hashmap *map, const char *key, void *value)
    31	{
    32		t_hashnode	new_node;
    33	
    34		if (!map || !key)
    35			return (-1);
    36		if (need_rehash(map) && hashmap_rehash(map) == -1)
    37			return (-1);
    38		if (prepare_new_node(&new_node, key, value) == -1)
    39			return (-1);
    40		return (robinhood_probe(map, new_node));
    41	}
    42	
    43	/***
    44	 * @brief Check if the hashmap needs to be rehashed
    45	 */
    46	static int	need_rehash(t_hashmap *map)
    47	{
    48		return (map->count * 10 >= map->capacity * 7);
    49	}
    50	
    51	/***
    52	 * @brief Prepare a new node for insertion
    53	 */
    54	static int	prepare_new_node(t_hashnode *node, const char *key, void *value)
    55	{
    56		node->key = ft_strdup(key);
    57		if (!node->key)
    58			return (-1);
    59		node->value = value;
    60		node->distance = 0;
    61		return (0);
    62	}
    63	
    64	/***
    65	 * @brief Swap two nodes
    66	 */
    67	static void	swap_nodes(t_hashnode *a, t_hashnode *b)
    68	{
    69		t_hashnode	tmp;
    70	
    71		tmp = *a;
    72		*a = *b;
    73		*b = tmp;
    74	}
    75	
    76	/***
    77	 * @brief Insert a new node into the hashmap using Robin Hood hashing
    78	 * @param map the hashmap to insert into
    79	 * @param new_node the new node to insert
    80	 * @return 0 on success, -1 on failure
    81	 */
    82	int	robinhood_probe(t_hashmap *map, t_hashnode new_node)
    83	{
    84		size_t	index;
    85	
    86		index = hash_function(new_node.key, map->capacity);
    87		while (map->nodes[index].key)
    88		{
    89			if (ft_strcmp(map->nodes[index].key, new_node.key) == 0)
    90			{
    91				free(map->nodes[index].key);
    92				map->nodes[index] = new_node;
    93				return (0);
    94			}
    95			if (new_node.distance > map->nodes[index].distance)
    96			{
    97				swap_nodes(&new_node, &map->nodes[index]);
    98			}
    99			index = (index + 1) % map->capacity;
   100			new_node.distance++;
   101		}
   102		map->nodes[index] = new_node;
   103		map->count++;
   104		return (0);
   105	}

    23	 * @brief Rehash the hashmap when the load factor is too high.
    24	 * @param map The hashmap to rehash.
    25	 * @return 0 on success, -1 on failure.
    26	 */
    27	int	hashmap_rehash(t_hashmap *map)
    28	{
    29		t_hashnode	*old_nodes;
    30		size_t		old_capacity;
    31	
    32		old_nodes = map->nodes;
    33		old_capacity = map->capacity;
    34		if (allocate_new_table(map) == -1)
    35		{
    36			map->nodes = old_nodes;
    37			return (-1);
    38		}
    39		reinsert_old_nodes(map, old_nodes, old_capacity);
    40		free(old_nodes);
    41		return (0);
    42	}
    43	
    44	/***
    45	 * @brief Allocate a new table with double the previous capacity.
    46	 * @param map The hashmap to update.
    47	 * @return 0 on success, -1 on failure.
    48	 */
    49	static int	allocate_new_table(t_hashmap *map)
    50	{
    51		t_hashnode	*new_nodes;
    52	
    53		new_nodes = (t_hashnode *)ft_calloc(map->capacity * 2, sizeof(t_hashnode));
    54		if (!new_nodes)
    55			return (-1);
    56		map->nodes = new_nodes;
    57		map->capacity *= 2;
    58		map->count = 0;
    59		return (0);
    60	}
    61	
    62	/***
    63	 * @brief Reinsert old nodes into the new table.
    64	 * @param map The new hashmap.
    65	 * @param old_nodes The old array of nodes.
    66	 * @param old_capacity The capacity of the old array.
    67	 */
    68	static void	reinsert_old_nodes(t_hashmap *map, t_hashnode *old_nodes,
    69		size_t old_capacity)
    70	{
    71		size_t		i;
    72		t_hashnode	new_node;
    73	
    74		i = 0;
    75		while (i < old_capacity)
    76		{
    77			if (old_nodes[i].key)
    78			{
    79				new_node = old_nodes[i];
    80				new_node.distance = 0;
    81				robinhood_probe(map, new_node);
    82			}
    83			i++;
    84		}
    85	}
